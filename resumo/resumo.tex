\begin{abstract}
\nota{Escrever por último, 150--200 palavras. Uma frase para cada:
(1) problema: backing track de percussão de samba para praticar improvisação no cavaquinho;
(2) método: gramática livre de contexto probabilística cujos terminais são ciclos de 2 compassos;
(3) restrição estilística: samba em 2/4, telecoteco, frases de 8 e blocos de 16 compassos marcados por virada ou breque;
(4) saída: MIDI reprodutível por seed, áudio com amostras recortadas do BRID;
(5) resultado: músicas diferentes por seed e por hiperparâmetro, e o que se ouve de diferente;
(6) a ressalva honesta: a linguagem gerada é regular, e a gramática se justifica pela legibilidade.}
\end{abstract}

\section{Introdução}\label{sec:introducao}

\nota{§1 Motivação. Quem improvisa no cavaquinho precisa de uma levada estável (a âncora),
que varie o suficiente para não soar como loop e que \emph{sinalize a forma}, para que ele
saiba onde está. Referência de ``feel'': playbacks de estudo de surdo, pandeiro e tamborim a
90 BPM, como os do YouTube.}

\nota{§2 Restrição estilística (exigida pelo TP, deixe explícita): samba em 2/4; ciclo
rítmico de 2 compassos ancorado no telecoteco do tamborim \cite{sandroni2001}; frases de 8
compassos agrupadas em blocos de 16, com virada ou breque no fim de cada bloco.
Instrumentos: surdos de primeira e segunda, pandeiro, tamborim e cuíca.}

\nota{§3 Método e contribuição em 2--3 frases: gramática probabilística em que a hierarquia
(música $\to$ bloco $\to$ frase $\to$ ciclo) é explícita nas regras, e as probabilidades viram
hiperparâmetros interpretáveis (quanto varia, quanto dura, quanto a cuíca aparece).}

\subsection{Representação}\label{sec:representacao}

\nota{Grade de semicolcheias: 8 passos por compasso de 2/4, 16 por ciclo. Cada uma das 8
pistas (Figura~\ref{fig:grade}) guarda, por passo, 0 (silêncio), 1 (batida) ou 2 (acento).
Justifique o terminal ser um \emph{ciclo de 2 compassos}, não um compasso: o telecoteco só se
completa em 2 compassos (``pergunta e resposta''), e um terminal de 1 compasso quebraria a
clave. Duração do ciclo: $16\Delta$, com $\Delta = 60/(4\cdot\mathrm{BPM})$, ou seja,
2,67\,s a 90 BPM.}

\subsection{Gramática}\label{sec:gramatica}

\nota{1 parágrafo apresentando a Tabela~\ref{tab:gramatica}: axioma \nt{Música}; os
não-terminais; os 14 terminais; cada lado esquerdo soma 1. Diga o que cada nível
significa musicalmente: \nt{Bloco} = 16 compassos = duas frases; \nt{Fim\_A} e \nt{Fim\_B}
são os fins de frase, onde caem viradas e breques (nunca no meio, para não desorientar quem
improvisa); \nt{G} é um ciclo comum, que varia no máximo um instrumento por vez.
Mencione o limite \texttt{max\_blocos} = 10, que \emph{trunca} a recursão de \nt{Blocos}.}

\subsection{Hiperparâmetros}\label{sec:hiper}

\nota{1 parágrafo curto sobre a Tabela~\ref{tab:hiper}. Destaque $\varepsilon$ como o
principal (de playback monótono a percussão ``conversando'') e $c$ e $v$ como o controle
independente da cuíca. Use a equação \eqref{eq:tamanho}: com $p=0{,}8$ e o truncamento em 10, $\mathbb{E}[n] = (1-p^{10})/(1-p)
\approx 4{,}46$ blocos, ou seja, cerca de 38 ciclos (aprox.\ 1min40 a 90 BPM).
O andamento (BPM) não é hiperparâmetro do método, só do render.}

\begin{table}[t]
  \centering
  \small
  \begin{tabular}{@{}clcl@{}}
    \toprule
    símb.         & flag                        & padrão & efeito \\
    \midrule
    $\varepsilon$ & \texttt{--epsilon}          & 0,15   & variação de tamborim e pandeiro \\
    $c$           & \texttt{--cuica-presenca}   & 0,25   & frequência das frases de cuíca \\
    $v$           & \texttt{--cuica-variedade}  & 0,5    & 0: sempre \tm{cu1}; 1: 4 equiprováveis \\
    $p$           & \texttt{--p}                & 0,8    & continuação (tamanho) \\
    $v_a$         & \texttt{--va}               & 0,3    & virada no meio do bloco \\
    $b$           & \texttt{--b}                & 0,3    & breque no fim do bloco \\
    \bottomrule
  \end{tabular}
  \caption{Hiperparâmetros da gramática ($\varepsilon + c \le 1$).}
  \label{tab:hiper}
\end{table}

\begin{equation}\label{eq:tamanho}
  \mathbb{E}[n] = \sum_{k=0}^{n_{\max}-1} p^k = \frac{1-p^{\,n_{\max}}}{1-p},
  \qquad \text{ciclos} = 2 + 8n
\end{equation}


\subsection{Do símbolo ao áudio}\label{sec:audio}

\nota{§1 MIDI. Cada nota cai em $t = (16c + i)\Delta + \mathcal{N}(0,\sigma)$, com
$\sigma = 4$\,ms de humanização; velocidade 78 (batida) ou 108 (acento) mais
$\mathcal{N}(0,6)$. Um instrumento MIDI por grupo (surdo, pandeiro, tamborim, cuíca), com
notas do mapa General MIDI. Um \texttt{.json} ao lado guarda a seed, os hiperparâmetros e a
derivação. A gramática, a humanização e o sampler usam geradores com seeds distintas
(\textit{seed}, +1, +2): trocar o render não muda a música, e a mesma seed gera o mesmo
\texttt{.mid}.}

\nota{§2 Sampler BRID \cite{maia2018} (CC BY 4.0). Faixas solo de samba do mesmo
instrumentista; recorte por onset (librosa); como os solos não têm anotação de beat, a grade
é ajustada aos próprios onsets; rótulos de articulação por k-means (pandeiro, $k=3$), por
$f_0$ (cuíca: grave e aguda) e por decaimento (surdo aberto); 2 camadas de dinâmica por RMS;
6 recortes por camada, os mais típicos; round-robin sem repetição. Fallback: FluidSynth com
soundfont GM. Mantenha curto: é infraestrutura, não o método.}

\section{Discussão e Trabalhos Futuros}\label{sec:discussao}

\nota{§1 Expressividade, a ressalva central: a linguagem gerada é \textbf{regular}
(\nt{Blocos} só é recursivo à direita e o resto é finito), então uma cadeia de Markov com
estados de \emph{posição na frase} geraria as mesmas músicas. O argumento a favor da gramática
é legibilidade e hierarquia explícita: \nt{Fim\_A} e \nt{Fim\_B} \emph{são} esses estados,
mas com nome e significado musical. Não diga que a gramática é ``mais poderosa''.}

\nota{§2 Limitações: (a) o surdo é fixo porque uma gramática livre de contexto não ``lembra''
uma escolha feita no início; seria preciso duplicar regras ou usar uma gramática de
atributos; pela mesma razão, a cuíca pode repetir a mesma frase em ciclos seguidos; (b) é
variação combinatória sobre um vocabulário fixo, não improvisação; (c) os padrões foram
escritos à mão; (d) as amostras vêm de instrumentistas e salas diferentes; (e) a grade é
quantizada, sem o microtiming (``balanço'') do samba.}

\nota{§3 Trabalhos futuros: gramática de atributos (surdo e motivo da cuíca por música);
variação condicionada à harmonia do cavaquinho; caixa e repique; microtiming medido no BRID;
transcrever o pandeiro a partir dos dados.}
